### apostrophe
# PowerShell with the built-in UI Automation assemblies: https://learn.microsoft.com/dotnet/api/system.windows.automation.automationelement
# Run it in Windows PowerShell with the -STA switch (powershell.exe -STA -File script.ps1).
Add-Type -AssemblyName UIAutomationClient
Add-Type -AssemblyName UIAutomationTypes

$root = [System.Windows.Automation.AutomationElement]::RootElement
$windowCondition = New-Object System.Windows.Automation.PropertyCondition([System.Windows.Automation.AutomationElement]::ProcessIdProperty, 4242)  # the process id changes on every run; match the window name instead if needed
$window = $root.FindFirst([System.Windows.Automation.TreeScope]::Children, $windowCondition)
$c1 = New-Object System.Windows.Automation.PropertyCondition([System.Windows.Automation.AutomationElement]::NameProperty, 'Don''t allow')
$element = $window.FindFirst([System.Windows.Automation.TreeScope]::Descendants, $c1)
# Invoke works for buttons and similar controls; other controls need another pattern.
$element.GetCurrentPattern([System.Windows.Automation.InvokePattern]::Pattern).Invoke()
