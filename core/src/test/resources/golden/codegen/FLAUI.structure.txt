### automation id
// FlaUI: https://github.com/FlaUI/FlaUI/wiki
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

using var app = Application.Attach(4242);  // the process id changes on every run; Attach("<exe name>") is stable
using var automation = new UIA3Automation();
var window = app.GetMainWindow(automation);
var element = window.FindFirstDescendant(cf => cf.ByAutomationId("num1Button"));
element.Click();
